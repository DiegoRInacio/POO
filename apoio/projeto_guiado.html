Projeto Guiado: da Análise ao Código

Sistema de Biblioteca construído passo a passo. O objetivo não é apenas chegar ao código, mas tornar explícitas as decisões que ligam o problema, o diagrama e os testes.

1Entender o problema

Uma biblioteca possui livros catalogados. Cada livro pode ter exemplares físicos. Usuários realizam empréstimos de exemplares disponíveis e depois os devolvem.

Regras mínimas: um exemplar pertence a um livro; exemplar emprestado não pode ser emprestado novamente; um usuário mantém seu histórico de empréstimos; devolver encerra o empréstimo e libera o exemplar.

2Identificar objetos do domínio

ObjetoResponsabilidadeEstado essencial
LivroRepresentar a obra catalogadatítulo, ISBN, exemplares
ExemplarRepresentar uma cópia físicacódigo, disponibilidade
UsuarioRepresentar quem toma emprestadonome, empréstimos
EmprestimoRegistrar a operação e seu estadoexemplar, usuário, ativo
BibliotecaCoordenar catálogo e empréstimoslivros, empréstimos
Evite transformar verbos em classes sem necessidade: “emprestar” é uma operação coordenada pela biblioteca; o registro resultante é o objeto Emprestimo.

3Decidir os relacionamentos

RelaçãoDecisãoJustificativa
Livro–ExemplarComposiçãoNeste modelo didático, o exemplar é criado pelo livro e pertence ao seu catálogo.
Biblioteca–LivroAgregaçãoLivros são criados fora e cadastrados na biblioteca.
Usuário–EmpréstimoAgregaçãoO usuário referencia registros coordenados pela biblioteca.
Empréstimo–ExemplarAssociaçãoO empréstimo conhece o exemplar, mas não controla sua existência.

Essas escolhas pertencem a este recorte do domínio. Em outro sistema, persistência e regras institucionais poderiam levar a decisões diferentes.

5Implementação completa em Python

class ExemplarIndisponivelError(Exception):
    pass


class Exemplar:
    def __init__(self, codigo):
        self.codigo = codigo
        self.disponivel = True


class Livro:
    def __init__(self, titulo, isbn):
        self.titulo = titulo
        self.isbn = isbn
        self.exemplares = []

    def adicionar_exemplar(self, codigo):
        exemplar = Exemplar(codigo)
        self.exemplares.append(exemplar)
        return exemplar


class Usuario:
    def __init__(self, nome):
        self.nome = nome
        self.emprestimos = []


class Emprestimo:
    def __init__(self, exemplar, usuario):
        self.exemplar = exemplar
        self.usuario = usuario
        self.ativo = True

    def devolver(self):
        if self.ativo:
            self.ativo = False
            self.exemplar.disponivel = True


class Biblioteca:
    def __init__(self):
        self.livros = []
        self.emprestimos = []

    def cadastrar_livro(self, livro):
        self.livros.append(livro)

    def emprestar(self, exemplar, usuario):
        if not exemplar.disponivel:
            raise ExemplarIndisponivelError(
                "Exemplar indisponível"
            )
        exemplar.disponivel = False
        emprestimo = Emprestimo(exemplar, usuario)
        self.emprestimos.append(emprestimo)
        usuario.emprestimos.append(emprestimo)
        return emprestimo

6Implementação completa em JavaScript

class ExemplarIndisponivelError extends Error {}

class Exemplar {
  constructor(codigo) {
    this.codigo = codigo;
    this.disponivel = true;
  }
}

class Livro {
  constructor(titulo, isbn) {
    this.titulo = titulo;
    this.isbn = isbn;
    this.exemplares = [];
  }

  adicionarExemplar(codigo) {
    const exemplar = new Exemplar(codigo);
    this.exemplares.push(exemplar);
    return exemplar;
  }
}

class Usuario {
  constructor(nome) {
    this.nome = nome;
    this.emprestimos = [];
  }
}

class Emprestimo {
  constructor(exemplar, usuario) {
    this.exemplar = exemplar;
    this.usuario = usuario;
    this.ativo = true;
  }

  devolver() {
    if (this.ativo) {
      this.ativo = false;
      this.exemplar.disponivel = true;
    }
  }
}

class Biblioteca {
  constructor() {
    this.livros = [];
    this.emprestimos = [];
  }

  cadastrarLivro(livro) {
    this.livros.push(livro);
  }

  emprestar(exemplar, usuario) {
    if (!exemplar.disponivel) {
      throw new ExemplarIndisponivelError(
        "Exemplar indisponível"
      );
    }
    exemplar.disponivel = false;
    const emprestimo = new Emprestimo(exemplar, usuario);
    this.emprestimos.push(emprestimo);
    usuario.emprestimos.push(emprestimo);
    return emprestimo;
  }
}

7Provar as regras com testes

Python · unittest
import unittest

class TestBiblioteca(unittest.TestCase):
    def setUp(self):
        self.biblioteca = Biblioteca()
        self.livro = Livro("1984", "9780451524935")
        self.exemplar = self.livro.adicionar_exemplar("EX-01")
        self.usuario = Usuario("Ana")

    def test_emprestar_e_devolver(self):
        emprestimo = self.biblioteca.emprestar(
            self.exemplar, self.usuario
        )
        self.assertFalse(self.exemplar.disponivel)
        emprestimo.devolver()
        self.assertTrue(self.exemplar.disponivel)

    def test_impedir_emprestimo_duplicado(self):
        self.biblioteca.emprestar(self.exemplar, self.usuario)
        with self.assertRaises(ExemplarIndisponivelError):
            self.biblioteca.emprestar(self.exemplar, self.usuario)
JavaScript · Jest
describe("Biblioteca", () => {
  let biblioteca, exemplar, usuario;

  beforeEach(() => {
    biblioteca = new Biblioteca();
    const livro = new Livro("1984", "9780451524935");
    exemplar = livro.adicionarExemplar("EX-01");
    usuario = new Usuario("Ana");
  });

  test("empresta e devolve", () => {
    const emprestimo = biblioteca.emprestar(
      exemplar, usuario
    );
    expect(exemplar.disponivel).toBe(false);
    emprestimo.devolver();
    expect(exemplar.disponivel).toBe(true);
  });

  test("impede empréstimo duplicado", () => {
    biblioteca.emprestar(exemplar, usuario);
    expect(() =>
      biblioteca.emprestar(exemplar, usuario)
    ).toThrow(ExemplarIndisponivelError);
  });
});
